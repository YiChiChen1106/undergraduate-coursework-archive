//define function of TMP102
void configure_TMP102(){
    // pointer register设为01，指向configuration register
    ConfigRegisterTMP102[0] = 0x01; 
    ConfigRegisterTMP102[1] = 0x60; //0110 0000
    ConfigRegisterTMP102[2] = 0xA0; //1010 0000
    Temperature_reg_address[0] = 0x00;  

    // 将这三个bytes发送给TMP102
    // 效果是把两个configuration register的值分别设置为了0x60和0xA0
    // TMP102Address
    TMP102.write(TMP102Address<<1,ConfigRegisterTMP102,3); // send configuration information of 3 byte  
}

void temperature_detect(){
    remain = 0;
    TMP102.read(TMP102Address<<1, temp_register, 2);
    M = temp_register[0] << 4;
    L = temp_register[1] >> 4;
    Temperature = (M + L)*0.0625;
    printf("Temperature = %.2f \n\r",Temperature);
}
//end define temperature

//define  accelerate
void ADXL345_SPI_Initialise()
{
    ADXL345_cs=1;
    ADXL345.format(8,3);
    ADXL345.frequency(2000000);
    ADXL345_cs=0;
    ADXL345.write(0x31);
    ADXL345.write(0x0B);
    ADXL345_cs=1;
    ADXL345_cs=0;
    ADXL345.write(0x2D);
    ADXL345.write(0x08);
    ADXL345_cs=1; 
}
void accelerate_detection(){
    ADXL345_cs=0;
    ADXL345.write(0x80|0x40|0x32);
    for(int i=0; i<5; i++){
        buffer[i] = ADXL345.write(0x00);
    } 
    ADXL345_cs = 1;
    data[0] = buffer[1]<<8 | buffer[0];
    data[1] = buffer[3]<<8 | buffer[2];
    data[2] = buffer[5]<<8 | buffer[4];
    x=0.004*data[0];
    y=0.004*data[1];
    z=0.004*data[2];
    printf("x=%+1.2fg\t y=%+1.2fg\t z=%+1.2fg\n\r", x, y, z);
}
//end define acceleration

i++) {
 }

 TMP102.write(SSD1306_I2C_ADDRESS, data, 2);
 }
 void ssd1306_draw_char(int x, int y, char c) {
 if (c < ' ' || c > 'z') {
 c = ' '; // Replace unsupported characters with a space
 }
 c -= ' ';
 for (int i = 0; i < 5; i++) {
 char data[2];
data[0] = 0x40; 
data[1] = font[c * 5 + i];
 TMP102.write(SSD1306_I2C_ADDRESS, data, 2);
 }
 char space[2] = {0x40, 0x00};
TMP102.write(SSD1306_I2C_ADDRESS, space, 2);
 }
 void ssd1306_draw_string(int x, int y, const char *str) {
 while (*str) {
 ssd1306_draw_char(x, y, *str++);
 }
 }

//define buzzer
void alarm(){
    for(int i=0;i<=27;i++){
        //Change PWM's period to change the frequency of the music
        buzzer.period(1/(2*frequency1[i]));
        //duty cycle
        buzzer=0.5;
        //time interruption for beat
        wait_us(1000000*0.4*beat1[i]);
    }
}
//end define buzzer


int main()
    {
    // main() runs in its own thread in the OS
    configure_TMP102();
    ADXL345_SPI_Initialise();

    Temperature_reg_address[0] = 0x00 ; // point to temp register
    TMP102.write(TMP102Address<<1, Temperature_reg_address, 1); // send temp information of 1 byte  

    while (true) {
        temperature_detect();
        accelerate_detection();
        ssd1306_init();
        ssd1306_clear();
        ssd1306_draw_string(0, 0,"hang on...");
        if(Temperature > 28 | Temperature < 25 | x > 1.5 | y > 1.5 | z > 1.5 | x < -1.5 | y < -1.5 | z < -1.5){
            ssd1306_init();
            ssd1306_clear();
            ssd1306_draw_string(0, 0,"Warning ! Warning ! Warning !");
            break;
        }
        wait_us(1000000);
    }   
    alarm();
    return 0;
}
